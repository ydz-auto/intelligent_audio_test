# 29_设备驱动导轨控制集成

> 文件：`backend/device_driver/base_driver.py`、`backend/device_driver/driver_factory.py`

## 现状分析

### 现有驱动工厂

`driver_factory.py` 通过 `register_specialized_driver` 注册不同类型的设备驱动。当前支持的驱动类型包括 HarmonyOS、Android 等。

### 无导轨控制

现有框架中没有导轨控制相关接口。导轨是 E2E 测试中的物理设备，用于自动调整被测设备与采集麦克风的距离。

## 改造方案

### 设计原则

导轨控制集成到现有 `device_driver` 框架，**不单独创建 RailController**。原因：
- 导轨是设备测试环境的一部分，与被测设备绑定
- 复用现有驱动的生命周期管理（初始化/销毁）
- 复用现有驱动的日志和事件系统

### 2. 有导轨的驱动实现

```python
class RailEquippedDriver(BaseDeviceDriver):
    """带导轨的设备驱动（如特定硬件驱动）"""
    
    def __init__(self):
        super().__init__()
        self._rail_position = 0.0  # 当前位置（cm）
        self._rail_max_distance = 100.0  # 最大距离（cm）
        self._rail_speed = 10.0  # 移动速度（cm/s）

    def move_rail(self, distance_cm: float) -> bool:
        """移动导轨到指定距离"""
        try:
            distance_cm = max(0, min(distance_cm, self._rail_max_distance))
            self._log(content=f"导轨移动: {self._rail_position:.1f}cm → {distance_cm:.1f}cm")
            
            # 发送导轨控制指令
            # 具体实现取决于导轨硬件协议（串口/TCP/USB）
            self._send_rail_command(distance_cm)
            
            # 等待移动完成
            self._wait_rail_complete(distance_cm)
            
            self._rail_position = distance_cm
            return True
        except Exception as e:
            self._log(level='ERROR', content=f"导轨移动失败: {e}")
            return False

    def reset_rail(self) -> bool:
        """导轨复位到最远端"""
        try:
            self._log(content=f"导轨复位: {self._rail_position:.1f}cm → 0.0cm")
            self._send_rail_command(0.0)
            self._wait_rail_complete(0.0)
            self._rail_position = 0.0
            return True
        except Exception as e:
            self._log(level='ERROR', content=f"导轨复位失败: {e}")
            return False

    def get_rail_position(self) -> float:
        return self._rail_position

    def _send_rail_command(self, target_cm: float):
        """发送导轨控制指令（硬件协议相关）"""
        raise NotImplementedError("子类需实现具体导轨协议")

    def _wait_rail_complete(self, target_cm: float):
        """等待导轨移动完成"""
        raise NotImplementedError("子类需实现移动完成检测")
```

### 4. E2E 执行器调用方式

```python
# e2e_executor.py _execute_rounds（每轮从 algorithmParams 读取）
def _execute_rounds(self, task_id, rounds, device_info_list):
    for i, round_config in enumerate(rounds):
        algo_params = CaseParameterExtractor.convert_params_to_dict(
            round_config.get('algorithmParams', [])
        )
        saved_state = {}

        # 获取导轨距离参数（来自 round.algorithmParams，field_code=railDistance）
        rail_distance = algo_params.get('railDistance')
        if rail_distance is not None:
            rail_distance = float(rail_distance)
            for dev in device_info_list:
                driver = dev.get('driver')
                if driver and hasattr(driver, 'move_rail'):
                    try:
                        driver.move_rail(rail_distance)
                        self._log(content=f"导轨移动到 {rail_distance}cm")
                        saved_state['rail'] = True
                    except NotImplementedError:
                        self._log(content="设备不支持导轨控制，跳过")

        try:
            # ... 多轮循环执行 ...
            pass
        finally:
            # 安全复位（try/finally 保证）
            if saved_state.get('rail'):
                for dev in device_info_list:
                    driver = dev.get('driver')
                    if driver and hasattr(driver, 'reset_rail'):
                        try:
                            driver.reset_rail()
                            self._log(content="导轨复位完成")
                        except (NotImplementedError, Exception) as e:
                            self._log(level='WARN', content=f"导轨复位失败: {e}")
```

### 5. 导轨距离参数来源

导轨距离参数从 `round.algorithmParams` 读取（`field_code='railDistance'`），不再从 config 顶层读取：

```python
# 读取方式
algo_params = CaseParameterExtractor.convert_params_to_dict(round_config.get('algorithmParams', []))
rail_distance = algo_params.get('railDistance')  # 字符串，需转为 float
```

### 6. 异常中断时的安全复位

```python
# 确保任何异常情况下都能安全复位（从 algorithmParams 读取 railDistance）
algo_params = CaseParameterExtractor.convert_params_to_dict(round_config.get('algorithmParams', []))
rail_distance = algo_params.get('railDistance')
if rail_distance:
    rail_distance = float(rail_distance)
try:
    driver.move_rail(rail_distance)
    # ... 测试执行（可能抛出异常）...
except TaskStoppedException:
    self._log(content="任务被停止，正在复位导轨...")
except Exception as e:
    self._log(level='ERROR', content=f"执行异常: {e}，正在复位导轨...")
finally:
    driver.reset_rail()  # 无论如何都要复位
```
