# 14_轮次请求构建

> 文件：`backend/utils/api_executor.py`

## 功能说明

`_send_round_request()` 负责构建并发送单轮对话请求到 api_adapter_service，等待响应并返回轮次结果。

## 方法设计

```python
def _send_round_request(
    self,
    api: dict,
    session: SessionContext,
    round_number: int,
    round_config: dict,
    algorithm_params: dict
) -> dict:
    """发送单轮对话请求
    
    Args:
        api: API 配置信息
        session: 会话上下文管理器
        round_number: 当前轮次序号
        round_config: 本轮配置 {inputType, inputText, audioId}
        algorithm_params: 算法参数
        
    Returns:
        dict: 轮次结果 {roundNumber, input, output, latency, ...}
    """
    start_time = time.time()
    
    # 1. 构建输入内容
    if round_config.get('inputType') == 'text':
        input_content = {'text': round_config['inputText']}
    else:
        # 音频输入：上传音频获取 audio_path
        audio_path = self._get_audio_path(round_config['audioId'])
        input_content = {'audio_path': audio_path}
    
    # 2. 构建请求体
    request_body = {
        'task_type': 'voice_llm',
        'session_id': session.session_id,
        'round': round_number,
        'input': input_content,
        'context': session.get_context(),  # 历史对话上下文
        'vendor_config': {
            'api_url': api.get('apiUrl'),
            'headers': api.get('headers', {}),
            'timeout': session.session_timeout
        }
    }
    
    # 3. 发送到 api_adapter_service
    adapter_url = f"{self._adapter_base_url}/api/v1/tasks"
    response = requests.post(
        adapter_url,
        json=request_body,
        timeout=session.session_timeout
    )
    response.raise_for_status()
    task_result = response.json()
    
    # 4. 提取结果
    latency = time.time() - start_time
    return {
        'roundNumber': round_number,
        'input': input_text or (audios[0].get('audio_id', '') if audios else ''),
        'inputType': 'text' if input_text else 'audio',
        'output': task_result.get('output_content', ''),
        'output_audio_path': task_result.get('output_audio_path'),
        'latency': round(latency, 3),
        'response_metrics': task_result.get('response_metrics', {})
    }
```

## 错误处理

```python
# 超时处理
try:
    response = requests.post(adapter_url, json=request_body, timeout=timeout)
except requests.Timeout:
    self._log(level='WARN', content=f"第 {round_number} 轮请求超时 ({timeout}s)")
    return {
        'roundNumber': round_number,
        'input': input_text,
        'output': '',
        'latency': timeout,
        'error': 'timeout'
    }

# HTTP 错误处理
except requests.RequestException as e:
    self._log(level='ERROR', content=f"第 {round_number} 轮请求失败: {e}")
    return {
        'roundNumber': round_number,
        'input': input_text,
        'output': '',
        'latency': 0,
        'error': str(e)
    }
```
