# 35 — 数据迁移方案

> **所属步骤**：数据迁移  
> **改造类型**：新增  
> **涉及文件**：`backend/scripts/migrations/` 目录

---

## 背景

voice_llm 改造涉及多项数据库结构变更：

1. **TestCase 表**：新增 `test_type`、`related_case_id` 字段
2. **TestResultDimension 表**：新增 `round_number` 字段
3. **CaseAlgorithmParam 表**：新增 `scope` 字段
4. **现有用例拆分**：每个现有 voice_llm 用例拆分为 api + e2e 两条记录

---

## 改造内容

### 1. DDL 迁移脚本

```sql
-- migration_001_voice_llm_schema.sql

-- 1. TestCase 表新增字段
ALTER TABLE test_case
ADD COLUMN IF NOT EXISTS test_type VARCHAR(10) DEFAULT 'api',
ADD COLUMN IF NOT EXISTS related_case_id INTEGER REFERENCES test_case(id);

-- 创建索引
CREATE INDEX IF NOT EXISTS idx_test_case_test_type ON test_case(test_type);
CREATE INDEX IF NOT EXISTS idx_test_case_related ON test_case(related_case_id);

-- 2. TestResultDimension 表新增字段
ALTER TABLE test_result_dimension
ADD COLUMN IF NOT EXISTS round_number INTEGER DEFAULT 0;

-- 创建索引
CREATE INDEX IF NOT EXISTS idx_trd_round ON test_result_dimension(test_result_id, round_number);

-- 3. CaseAlgorithmParam 表新增字段
ALTER TABLE case_algorithm_param
ADD COLUMN IF NOT EXISTS scope VARCHAR(10) DEFAULT 'common';

-- 创建索引
CREATE INDEX IF NOT EXISTS idx_cap_scope ON case_algorithm_param(algorithm_type, scope);

-- 4. TaskCase 表 extra_data 已有 JSONB 字段，无需 DDL
-- round_progress 存储在 extra_data JSON 中
```

### 2. 现有用例拆分脚本

```python
# split_testcase_by_type.py
"""
将现有 voice_llm 用例拆分为 api + e2e 双记录。

执行条件：
- test_case.algorithm_type = 'voice_llm'
- test_case.test_type IS NULL 或 = 'api' (未拆分的旧数据)

拆分逻辑：
1. 复制原记录为 e2e 记录
2. 原记录标记为 api
3. 设置 related_case_id 互相引用
4. 分别设置各自的 config JSON
"""

import json
import sys
from datetime import datetime

VALID_SCOPES_API = {'api', 'common'}
VALID_SCOPES_E2E = {'e2e', 'common'}

def migrate(app):
    """主迁移函数"""
    with app.app_context():
        from backend.models.models import db, TestCase

        # 查找所有未拆分的 voice_llm 用例
        cases = TestCase.query.filter(
            TestCase.algorithm_type == 'voice_llm',
            (TestCase.test_type == None) | (TestCase.test_type == 'api')
        ).all()

        print(f'Found {len(cases)} voice_llm cases to split')

        migrated = 0
        for case in cases:
            if case.related_case_id:
                continue  # 已拆分

            # 1. 创建 e2e 记录
            e2e_case = TestCase(
                name=case.name,
                group_id=case.group_id,
                algorithm_type=case.algorithm_type,
                test_type='e2e',
                config=extract_e2e_config(case.config),
                created_at=datetime.now(),
            )
            db.session.add(e2e_case)
            db.session.flush()

            # 2. 更新原记录
            case.test_type = 'api'
            case.config = extract_api_config(case.config)
            case.related_case_id = e2e_case.id
            e2e_case.related_case_id = case.id

            migrated += 1
            print(f'  Split case {case.id} → api({case.id}) + e2e({e2e_case.id})')

        db.session.commit()
        print(f'Migration complete: {migrated} cases split')


def _filter_algorithm_params(params, valid_scopes):
    """按 scope 过滤 algorithmParams 数组"""
    if not params:
        return []
    return [p for p in params if p.get('scope', 'common') in valid_scopes]


def extract_api_config(full_config):
    """从完整 config 中提取 API 部分"""
    if isinstance(full_config, str):
        full_config = json.loads(full_config)

    api_config = {
        'rounds': [],
    }

    for round_cfg in full_config.get('rounds', []):
        api_round = dict(round_cfg)
        api_round['algorithmParams'] = _filter_algorithm_params(
            round_cfg.get('algorithmParams', []), VALID_SCOPES_API
        )
        api_config['rounds'].append(api_round)

    # 只保留 api 相关的音频
    audios = full_config.get('audios', [])
    api_config['audios'] = [a for a in audios if a.get('testType') in ('api', 'both')]

    return json.dumps(api_config, ensure_ascii=False)


def extract_e2e_config(full_config):
    """从完整 config 中提取 E2E 部分（camelCase 命名）"""
    if isinstance(full_config, str):
        full_config = json.loads(full_config)

    e2e_config = {
        'rounds': [],
        'voiceprintConfig': full_config.get('voiceprint_config', {}),
        'interferers': full_config.get('interferer_config', []),
        'railDistance': full_config.get('rail_distance'),
        'volumeLevel': full_config.get('volume_level'),
        'interruptionEnabled': full_config.get('interruption_enabled', False),
    }

    for round_cfg in full_config.get('rounds', []):
        e2e_round = dict(round_cfg)
        e2e_round['algorithmParams'] = _filter_algorithm_params(
            round_cfg.get('algorithmParams', []), VALID_SCOPES_E2E
        )
        e2e_config['rounds'].append(e2e_round)

    # 只保留 e2e 相关的音频
    audios = full_config.get('audios', [])
    e2e_config['audios'] = [a for a in audios if a.get('testType') in ('e2e', 'both')]

    return json.dumps(e2e_config, ensure_ascii=False)


if __name__ == '__main__':
    from flask import create_app
    app = create_app()
    migrate(app)
```

### 4. 现有单轮结果 round_number 迁移

```sql
-- migration_003_round_number_default.sql

-- 现有单轮结果的 round_number 默认值应为 0（不是 NULL）
-- 多轮结果的 per-round 维度：round_number = 0, 1, 2, ... (0-indexed)
-- 整体维度：round_number = NULL

-- 更新现有 voice_llm 单轮结果：round_number 设为 0
UPDATE test_result_dimension trd
SET round_number = 0
FROM test_result tr
JOIN test_case tc ON tr.test_case_id = tc.id
WHERE trd.test_result_id = tr.id
  AND tc.algorithm_type = 'voice_llm'
  AND trd.round_number IS NULL;
```

### 6. 验证检查

```python
def verify_migration(app):
    """验证迁移结果"""
    with app.app_context():
        from backend.models.models import db, TestCase, TestResultDimension

        # 检查 1：所有 voice_llm 用例都有 test_type
        missing_type = TestCase.query.filter(
            TestCase.algorithm_type == 'voice_llm',
            TestCase.test_type == None,
        ).count()
        assert missing_type == 0, f'{missing_type} cases missing test_type'

        # 检查 2：所有 related_case_id 双向引用正确
        pairs = TestCase.query.filter(
            TestCase.related_case_id != None
        ).all()
        for case in pairs:
            related = TestCase.query.get(case.related_case_id)
            assert related.related_case_id == case.id, \
                f'Broken reference: {case.id} → {case.related_case_id}'

        # 检查 3：api 和 e2e 数量匹配
        api_count = TestCase.query.filter_by(
            algorithm_type='voice_llm', test_type='api'
        ).count()
        e2e_count = TestCase.query.filter_by(
            algorithm_type='voice_llm', test_type='e2e'
        ).count()
        assert api_count == e2e_count, \
            f'Mismatch: {api_count} api vs {e2e_count} e2e'

        # 检查 4：现有单轮结果的 round_number 默认值为 0（不是 NULL）
        null_round = TestResultDimension.query.filter(
            TestResultDimension.round_number == None
        ).join(
            TestResult, TestResultDimension.test_result_id == TestResult.id
        ).join(
            TestCase, TestResult.test_case_id == TestCase.id
        ).filter(
            TestCase.algorithm_type == 'voice_llm'
        ).count()
        assert null_round == 0, f'{null_round} voice_llm dimensions still have NULL round_number'

        print('All verification checks passed!')
```
