# 测试任务执行服务拆分方案

## 1. 背景与目标

### 1.1 当前问题

当前系统为单体 Flask 应用，所有功能（路由、设备驱动、任务执行、评估计算）都在同一个进程中运行，存在以下问题：

| 问题 | 说明 |
|------|------|
| **耦合严重** | 设备驱动与 Flask HTTP Server 共用进程，设备通信阻塞 HTTP 响应 |
| **扩展性差** | 无法独立扩展任务执行能力 |
| **故障影响大** | 任务执行异常可能影响 API 响应 |
| **资源竞争** | 设备控制线程与 HTTP 处理线程竞争资源 |
| **部署不灵活** | 无法针对不同负载场景独立部署 |
| **API 测试与 E2E 测试耦合** | API 测试不需要设备，但与 E2E 测试共用进程 |

### 1.3 拆分目标

实现四层微服务架构：

| 层级 | 服务 | 职责 |
|------|------|------|
| **网关层** | API Gateway | HTTP 路由、认证、限流、WebSocket 代理 |
| **调度层** | Task Service | 任务调度、分发、进度管理、结果汇总 |
| **执行层** | E2E Test Service | E2E 测试（需要设备）、设备驱动、音频播放 |
| **执行层** | API Test Service | API 测试（不需要设备）、并发测试、健康监控 |
| **评估层** | Evaluation Service | 评估计算、指标计算、报告生成 |

### 4.2 服务发现模块

```python
# shared/utils/service_discovery.py

from shared.models.database import db
from shared.models.models import ServiceRegistry
from datetime import datetime, timedelta
import random

class ServiceDiscovery:
    """服务发现与负载均衡"""
    
    @staticmethod
    def get_available_task_service():
        """获取可用的 Task Service 实例"""
        services = ServiceRegistry.query.filter_by(
            type='task_service',
            status='online'
        ).all()
        
        if not services:
            raise Exception('No available task service')
        
        return random.choice(services)
    
    @staticmethod
    def get_e2e_test_service(lab_name=None):
        """获取 E2E Test Service 实例"""
        query = ServiceRegistry.query.filter_by(
            type='e2e_test_service',
            status='online'
        )
        
        services = query.all()
        
        if lab_name:
            for service in services:
                labs = (service.capabilities or {}).get('labs', [])
                if lab_name in labs:
                    return service
        
        if not services:
            raise Exception('No available E2E test service')
        
        return random.choice(services)
    
    @staticmethod
    def get_api_test_service():
        """获取 API Test Service 实例"""
        services = ServiceRegistry.query.filter_by(
            type='api_test_service',
            status='online'
        ).all()
        
        if not services:
            raise Exception('No available API test service')
        
        return random.choice(services)
    
    @staticmethod
    def get_evaluation_service():
        """获取 Evaluation Service 实例"""
        services = ServiceRegistry.query.filter_by(
            type='evaluation_service',
            status='online'
        ).all()
        
        if not services:
            raise Exception('No available evaluation service')
        
        return random.choice(services)
    
    @staticmethod
    def cleanup_stale_services(timeout_seconds=60):
        """清理超时未心跳的服务"""
        threshold = datetime.utcnow() - timedelta(seconds=timeout_seconds)
        stale = ServiceRegistry.query.filter(
            ServiceRegistry.last_heartbeat < threshold,
            ServiceRegistry.status == 'online'
        ).all()
        
        for service in stale:
            service.status = 'offline'
        
        db.session.commit()
        return len(stale)
```

### 4.3 服务管理 API

```python
# api_gateway/blueprints/service_bp.py

from flask import Blueprint, request, jsonify
from shared.models.database import db
from shared.models.models import ServiceRegistry
from datetime import datetime

service_bp = Blueprint('service', __name__, url_prefix='/api/v1/services')

@service_bp.route('', methods=['GET'])
def list_services():
    """获取所有服务实例"""
    service_type = request.args.get('type')
    query = ServiceRegistry.query
    if service_type:
        query = query.filter_by(type=service_type)
    services = query.all()
    return jsonify({
        'services': [s.to_dict() for s in services]
    })

@service_bp.route('', methods=['POST'])
def create_service():
    """注册新服务实例"""
    data = request.json
    service = ServiceRegistry(
        name=data['name'],
        type=data['type'],
        host=data['host'],
        port=data['port'],
        grpc_port=data.get('grpc_port'),
        capabilities=data.get('capabilities', {}),
        metadata=data.get('metadata', {}),
        status='offline'
    )
    db.session.add(service)
    db.session.commit()
    return jsonify(service.to_dict()), 201

@service_bp.route('/<int:service_id>', methods=['PUT'])
def update_service(service_id):
    """更新服务实例配置"""
    service = ServiceRegistry.query.get_or_404(service_id)
    data = request.json
    for key in ['name', 'host', 'port', 'grpc_port', 'capabilities', 'metadata']:
        if key in data:
            setattr(service, key, data[key])
    db.session.commit()
    return jsonify(service.to_dict())

@service_bp.route('/<int:service_id>', methods=['DELETE'])
def delete_service(service_id):
    """删除服务实例"""
    service = ServiceRegistry.query.get_or_404(service_id)
    db.session.delete(service)
    db.session.commit()
    return '', 204

@service_bp.route('/<int:service_id>/heartbeat', methods=['POST'])
def heartbeat(service_id):
    """服务心跳"""
```

### 5.4 改造后的 ExecutionEngine

```python
# task_service/core/execution_engine.py

from shared.clients.e2e_test_client import E2ETestClient
from shared.clients.api_test_client import APITestClient
from shared.utils.service_discovery import ServiceDiscovery
from shared.models.database import db
from shared.models.models import Task

class ExecutionEngine:
    def __init__(self):
        self.workers = {}
        self.stop_flags = {}
        self.pause_flags = {}
        self.event_manager = EventManager(self)
        
        self.service_discovery = ServiceDiscovery()
        self._e2e_clients = {}
        self._api_test_clients = {}
    
    def _get_e2e_client(self, lab_name=None):
        """获取 E2E Test Service 客户端"""
        service = self.service_discovery.get_e2e_test_service(lab_name)
        cache_key = f"{service.host}:{service.grpc_port}"
        if cache_key not in self._e2e_clients:
            self._e2e_clients[cache_key] = E2ETestClient(service.host, service.grpc_port)
        return self._e2e_clients[cache_key]
    
    def _get_api_test_client(self):
        """获取 API Test Service 客户端"""
        service = self.service_discovery.get_api_test_service()
        cache_key = f"{service.host}:{service.port}"
        if cache_key not in self._api_test_clients:
            self._api_test_clients[cache_key] = APITestClient(service.host, service.port)
        return self._api_test_clients[cache_key]
    
    def start_task(self, task_id):
        """启动任务 - 分发到对应执行服务"""
        task = Task.query.get(task_id)
        if not task:
            raise Exception(f'Task {task_id} not found')
        
        case_ids = [tc.test_case_id for tc in task.task_cases]
        
        if task.type == 'e2e':
            client = self._get_e2e_client(lab_name=task.lab_name)
            device_id = task.device_id
            response = client.start_task(task_id, case_ids, device_id)
            self._log(task_id, 'INFO', f'E2E task started: {response}')
        else:
            client = self._get_api_test_client()
            api_ids = [api.id for api in task.apis]
            response = client.start_task(task_id, case_ids, api_ids)
            self._log(task_id, 'INFO', f'API test task started: {response}')
    
    def stop_task(self, task_id):
        """停止任务"""
        task = Task.query.get(task_id)
        if not task:
            return
        
        if task.type == 'e2e':
            client = self._get_e2e_client()
            client.stop_task(task_id)
        else:
            client = self._get_api_test_client()
            client.stop_task(task_id)
    
    def get_task_status(self, task_id):
        """获取任务状态"""
        task = Task.query.get(task_id)
        if not task:
            return None
        
        if task.type == 'e2e':
            client = self._get_e2e_client()
            return client.get_task_status(task_id)
        else:
            client = self._get_api_test_client()
            return client.get_task_status(task_id)
```

```python
# shared/utils/redis_pubsub.py

import redis
import json
import os

class RedisPubSub:
    _instance = None
    
    def __new__(cls, redis_url=None):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            redis_url = redis_url or os.environ.get('REDIS_URL', 'redis://localhost:6379')
            cls._instance.redis_client = redis.from_url(redis_url)
        return cls._instance
    
    def publish(self, channel, message):
        self.redis_client.publish(channel, json.dumps(message, ensure_ascii=False))
    
    def publish_progress(self, task_id, progress_data):
        self.publish('task_progress', {
            'task_id': task_id,
            **progress_data
        })
    
    def publish_log(self, task_id, log_data):
        self.publish('task_logs', {
            'task_id': task_id,
            **log_data
        })
    
    def publish_device_status(self, device_id, status_data):
        self.publish('device_status', {
            'device_id': device_id,
            **status_data
        })
    
    def subscribe(self, channels, callback):
        pubsub = self.redis_client.pubsub()
        pubsub.subscribe(channels)
        for message in pubsub.listen():
            if message['type'] == 'message':
                channel = message['channel']
                if isinstance(channel, bytes):
                    channel = channel.decode('utf-8')
                data = json.loads(message['data'])
                callback(channel, data)
```

#### API Gateway WebSocket Hub

```python
# api_gateway/websocket/hub.py

import threading
import json
import redis
import os

class WebSocketHub:
    def __init__(self, socketio, redis_url=None):
        self.socketio = socketio
        redis_url = redis_url or os.environ.get('REDIS_URL', 'redis://localhost:6379')
        self.redis_client = redis.from_url(redis_url)
        self.pubsub = self.redis_client.pubsub()
        self._running = False
    
    def start(self):
        self._running = True
        self.pubsub.subscribe('task_progress', 'task_logs', 'device_status')
        
        thread = threading.Thread(target=self._listen_loop, daemon=True)
        thread.start()
        print("[WebSocketHub] Started listening for events")
    
    def _listen_loop(self):
        for message in self.pubsub.listen():
            if not self._running:
                break
            
            if message['type'] == 'message':
                channel = message['channel']
                if isinstance(channel, bytes):
                    channel = channel.decode('utf-8')
                
                data = json.loads(message['data'])
                
                if channel == 'task_progress':
                    self.socketio.emit('task_progress', data, room=f"task_{data['task_id']}")
                elif channel == 'task_logs':
                    self.socketio.emit('logs', {
                        'type': 'LOG_BATCH',
                        'data': [data]
                    }, namespace='/ws/logs')
                    if data.get('task_id'):
                        self.socketio.emit('task_log', {
                            'taskId': data['task_id'],
                            'log': data
                        })
                elif channel == 'device_status':
                    self.socketio.emit('device_status', data, room=f"device_{data['device_id']}")
    
    def stop(self):
        self._running = False
        self.pubsub.unsubscribe()
```

```python
def log_and_emit(level, module, content, service_name='backend', 
                 category='system', source='backend', task_id=None, 
                 device_id=None, api_id=None, test_case_id=None, 
                 algorithm_type=None, push_to_websocket=True,

            args=(),
            exc_info=None

        record.service_name = service_name
        record.push_to_websocket = push_to_websocket
        record.from_log_and_emit = True
        
        for key, value in kwargs.items():
            setattr(record, key, value)

            record.thread_id = kwargs.get('thread_id') or str(threading.get_ident())
        
        handler = get_db_handler(service_name)
        handler.emit(record)
        
    except Exception as e:

class DatabaseLogHandler(logging.Handler):
    
    def __init__(self, service_name='backend'):
        super().__init__()

        self.recent_logs = {}

        self.enable_console_log = enable
    
    def _get_session(self):
        if self._SessionLocal is None:
            try:
                engine = create_engine(self._db_url, pool_size=5, pool_pre_ping=True)
                self._SessionLocal = sessionmaker(bind=engine)
            except Exception as e:
                print(f"[{self.service_name}] Failed to create session: {e}")
                return None
        return self._SessionLocal()
    
    def _get_log_model(self):
        if self._Log is None:
            try:
                from shared.models.models import Log
                self._Log = Log
            except ImportError:
                try:
                    from backend.models.models import Log
                    self._Log = Log
                except ImportError:
                    print(f"[{self.service_name}] Failed to import Log model")
                    return None
        return self._Log
    
    def _worker(self):
        batch = []
        last_flush = time.time()
        
        print(f"[{datetime.now(timezone(timedelta(hours=8))).strftime('%Y-%m-%d %H:%M:%S')}] - {self.service_name} - log_worker - INFO - Worker thread started.")
        
        while True:
            try:
                try:
                    data = self.queue.get(timeout=0.1)
                    if data is None:
                        break
                    batch.append(data)
                except queue.Empty:
                    pass
                
                current_time = time.time()
                should_flush = (
                    len(batch) >= self._batch_size or
                    (batch and (current_time - last_flush) >= self._batch_timeout)
                )
                
                if should_flush and batch:
                    self._process_batch(batch)
                    
                    for data in batch:
                        if data.get('push_to_websocket'):
                            self._emit_to_redis(data)
                    
                    batch = []
                    last_flush = current_time
                    
                    if current_time - self._last_archive_check >= self._archive_check_interval:
                        self._check_and_archive()
                        self._last_archive_check = current_time
                
            except Exception as e:
                print(f"[{datetime.now(timezone(timedelta(hours=8))).strftime('%Y-%m-%d %H:%M:%S')}] - {self.service_name} - log_worker - CRITICAL ERROR - {str(e)}")
                time.sleep(0.5)
    
    def _process_batch(self, batch):
        Log = self._get_log_model()
        if not Log:
            return
        
        session = self._get_session()
        if not session:
            return
        
        try:
            for data in batch:
                log_entry = Log(
                    time=data['time'],
                    level=data['level'],
                    category=data['category'],
                    module=data['module'],
                    source=data['source'],
                    content=data['content'],
                    task_id=data.get('task_id'),
                    device_id=data.get('device_id'),
                    api_id=data.get('api_id'),
                    test_case_id=data.get('test_case_id'),
                    thread_id=data.get('thread_id'),
                    algorithm_type=data.get('algorithm_type'),
                    service_name=data.get('service_name', self.service_name),
                )
                session.add(log_entry)
            session.commit()
        except Exception as e:
            session.rollback()
            print(f"[{datetime.now(timezone(timedelta(hours=8))).strftime('%Y-%m-%d %H:%M:%S')}] - {self.service_name} - log_worker - DB ERROR - {str(e)}")
        finally:
            session.close()

            redis_client = get_redis_client(self._redis_url)
            
            log_payload = {
                'time': data['time'].isoformat() if isinstance(data['time'], datetime) else data['time'],
                'level': data['level'],
                'module': data['module'],
                'content': data['content'],
                'service_name': data.get('service_name', self.service_name),
                'task_id': data.get('task_id'),
            }
            
            redis_client.publish('task_logs', json.dumps(log_payload, ensure_ascii=False))
            
        except Exception as e:
            print(f"[{datetime.now(timezone(timedelta(hours=8))).strftime('%Y-%m-%d %H:%M:%S')}] - {self.service_name} - log_worker - REDIS ERROR - {str(e)}")
    
    def _check_and_archive(self):
        pass
    
    def emit(self, record):
        try:
            if hasattr(record, 'module') and record.module in ['log_controller', 'log_handler', 'database']:
                return
            
            log_message = self.format(record)
            log_message = re.sub(r'^\[\d{4}-\d{2}-\d{2}T[\d\.:+\-]+\]', '', log_message).strip()
            
            if 'WebSocket' in log_message or 'socketio' in log_message:
                return
            
            log_fingerprint = hashlib.md5(f"{record.levelno}-{record.module}-{log_message}".encode('utf-8')).hexdigest()
            current_time = datetime.now().timestamp()
            
            if log_fingerprint in self.recent_logs:
                if current_time - self.recent_logs[log_fingerprint] < self.log_ttl:
                    return
            
            self.recent_logs[log_fingerprint] = current_time
            
            if len(self.recent_logs) > self.max_recent_logs:
                self.recent_logs = {fp: ts for fp, ts in self.recent_logs.items() if current_time - ts < self.log_ttl}
            
            log_data = {
                'time': datetime.now(timezone(timedelta(hours=8))),
                'level': record.levelname.upper(),
                'module': record.module if hasattr(record, 'module') else 'unknown',
                'category': getattr(record, 'category', 'system').lower(),
                'source': getattr(record, 'source', 'backend').lower(),
                'content': log_message,
                'task_id': getattr(record, 'task_id', None),
                'device_id': getattr(record, 'device_id', None),
                'api_id': getattr(record, 'api_id', None),
                'test_case_id': getattr(record, 'test_case_id', None),
                'thread_id': getattr(record, 'thread_id', None) or str(threading.get_ident()),
                'algorithm_type': getattr(record, 'algorithm_type', None),
                'service_name': getattr(record, 'service_name', self.service_name),
                'push_to_websocket': getattr(record, 'push_to_websocket', True),
            }
            
            self.queue.put(log_data)
            
        except Exception as e:
            if self.enable_console_log:
                print(f"Error putting log into queue: {str(e)}", file=sys.stderr)
```

```python
from typing import Dict, List, Any, Optional, Callable
from threading import Lock
from shared.algorithm.algorithm_config_loader import get_config_loader
from shared.utils.log_handler import log_not_emit


class FieldMapper:
    _instance = None
    _instance_lock = Lock()
    _config_lock = Lock()

    def __new__(cls):
        if cls._instance is None:
            with cls._instance_lock:
                if cls._instance is None:
                    cls._instance = super().__new__(cls)
                    cls._instance._initialized = False
        return cls._instance

    def __init__(self):
        if self._initialized:
            return
        self._initialized = True
        self._loader = get_config_loader()
        self._field_cache: Dict[str, Dict[str, Any]] = {}
        self._transforms: Dict[str, Callable] = {}
        self._register_builtin_transforms()
        log_not_emit('DEBUG', 'field_mapper', 'FieldMapper initialized', 
                     service_name='shared', category='algorithm')

    def _register_builtin_transforms(self):
        self._transforms = {
            'none': lambda x: x,
            'json_parse': self._json_parse,
            'base64': self._base64_encode,
            'to_string': lambda x: str(x) if x is not None else '',
            'to_int': lambda x: int(x) if x is not None else 0,
            'to_float': lambda x: float(x) if x is not None else 0.0,
            'to_bool': lambda x: bool(x) if x is not None else False,
            'rttm_to_obj': self._rttm_to_obj,
            'stm_to_obj': self._stm_to_obj,
        }

    def get_field_definitions(self, algorithm_type: str, field_type: str) -> Dict[str, Any]:
        return self._loader.get_field_definitions(algorithm_type, field_type)

    def transform_value(self, value: Any, transform_name: str) -> Any:
        transform_func = self._transforms.get(transform_name)
        if transform_func:
            return transform_func(value)
        return value

    def build_api_request(self, algorithm_type: str, params: Dict) -> Dict:
        pass
```

```python
# shared/algorithm/algorithm_config_loader.py

from typing import Dict, List, Any, Optional
from datetime import datetime
from threading import Lock
from shared.models.algorithm_models import (
    AlgorithmDefinition,
    AlgorithmDeviceParam,
    AlgorithmApiParam,
    EvaluationDimensionParam,
    ParamMapping,
    CaseAlgorithmParam,
    AlgorithmReferenceParam
)
from shared.utils.log_handler import log_not_emit


class AlgorithmConfigLoader:
    _instance = None
    _instance_lock = Lock()
    _config_cache: Dict[str, Any] = {}
    _last_reload_time: Optional[datetime] = None
    _reload_lock = Lock()

    def __new__(cls):
        if cls._instance is None:
            with cls._instance_lock:
                if cls._instance is None:
                    cls._instance = super().__new__(cls)
                    cls._instance._load_all_configs()
        return cls._instance

    def _load_all_configs(self):
        with self._reload_lock:
            log_not_emit('DEBUG', 'algorithm_config_loader', 
                        'Starting to load all algorithm configs', 
                        service_name='shared', category='algorithm')
            self._config_cache = {
                'algorithms': {},
                'device_params': {},
                'api_params': {},
                'case_params': {},
                'evaluation_dimension_params': {},
                'mappings': {},
                'reference_params': {}
            }
            
            algorithms = AlgorithmDefinition.query.filter_by(
                status='online', deleted=False
            ).all()
            
            for algo in algorithms:
                algo_type = algo.type
                self._config_cache['algorithms'][algo_type] = {
                    'id': algo.id,
                    'type': algo.type,
                    'name': algo.name,
                }

    def get_algorithm(self, algorithm_type: str) -> Optional[Dict]:
        return self._config_cache['algorithms'].get(algorithm_type)

    def get_device_params(self, algorithm_type: str) -> List[Dict]:
        return self._config_cache['device_params'].get(algorithm_type, [])

    def get_api_params(self, algorithm_type: str) -> List[Dict]:
        return self._config_cache['api_params'].get(algorithm_type, [])

    def reload_configs(self):
        self._load_all_configs()


def get_config_loader() -> AlgorithmConfigLoader:
    return AlgorithmConfigLoader()
```

```python
# api_gateway/middleware/auth.py

from functools import wraps
from flask import request, jsonify, g
import jwt

def get_current_user_id():
    """从 JWT Token 获取当前用户ID"""
    token = request.headers.get('Authorization', '').replace('Bearer ', '')
    if not token:
        return None
    
    try:
        payload = jwt.decode(token, current_app.config['SECRET_KEY'], algorithms=['HS256'])
        return payload.get('user_id')
    except jwt.InvalidTokenError:
        return None


def require_user_ownership(model_class, id_param='id'):
    """
    装饰器：检查用户是否有权限访问资源
    
    用法：
    @require_user_ownership(TestCase, 'case_id')
    def get_test_case(case_id):
        ...
    """
    def decorator(f):
        @wraps(f)
        def decorated_function(*args, **kwargs):
            user_id = get_current_user_id()
            if not user_id:
                return jsonify({'error': 'Unauthorized'}), 401
            
            resource_id = kwargs.get(id_param) or request.view_args.get(id_param)
            if not resource_id:
                return jsonify({'error': 'Resource ID required'}), 400
            
            resource = model_class.query.get(resource_id)
            if not resource:
                return jsonify({'error': 'Resource not found'}), 404
            
            if resource.user_id != user_id:
                return jsonify({'error': 'Forbidden'}), 403
            
            g.current_resource = resource
            return f(*args, **kwargs)
        return decorated_function
    return decorator


def filter_by_user(query, model_class):
    """根据当前用户过滤查询结果"""
    user_id = get_current_user_id()
    if user_id and hasattr(model_class, 'user_id'):
        return query.filter(model_class.user_id == user_id)
    return query
```

### 12.9 日志用户ID处理方案

#### 方案对比

| 方案 | 优点 | 缺点 |
|------|------|------|
| **不添加 user_id** | 减少存储开销；通过关联表查询用户 | 查询需要 JOIN；权限控制复杂 |
| **添加 user_id** | 直接按用户过滤；权限控制简单 | 增加存储开销；写入时需要获取 user_id |

#### 建议：添加 user_id

原因：
1. 日志查询时可以直接按用户过滤，不需要 JOIN
2. 用户只能看到自己相关的日志
3. 数据隔离更彻底
4. 日志量虽大，但 user_id 只是 8 字节的 BIGINT

```python
def log_and_emit(level, module, content, service_name='backend',

def _resolve_user_id(task_id=None, device_id=None, api_id=None):
    """从关联资源获取 user_id"""
    if task_id:
        try:
            from shared.models.models import TestTask
            task = TestTask.query.get(task_id)
            if task and task.user_id:
                return task.user_id
        except:
            pass
    
    if device_id:
        try:
            from shared.models.models import Device
            device = Device.query.get(device_id)
            if device and device.user_id:
                return device.user_id
        except:
            pass
    
    if api_id:
        try:
            from shared.models.models import API
            api = API.query.get(api_id)
            if api and api.user_id:
                return api.user_id
        except:
            pass
    
    return None


class DatabaseLogHandler(logging.Handler):
    
    def _process_batch(self, batch):
        Log = self._get_log_model()
        if not Log:
            return
        
        session = self._get_session()
        if not session:
            return
        
        try:
            for data in batch:
                log_entry = Log(
                    time=data['time'],
                    level=data['level'],
                    category=data['category'],
                    module=data['module'],
                    source=data['source'],
                    content=data['content'],
                    task_id=data.get('task_id'),
                    device_id=data.get('device_id'),
                    api_id=data.get('api_id'),
                    test_case_id=data.get('test_case_id'),
                    thread_id=data.get('thread_id'),
                    algorithm_type=data.get('algorithm_type'),
                    service_name=data.get('service_name', self.service_name),
                    user_id=data.get('user_id'),  # 新增
                )
                session.add(log_entry)
            session.commit()
        except Exception as e:
            session.rollback()
            print(f"[{datetime.now(timezone(timedelta(hours=8))).strftime('%Y-%m-%d %H:%M:%S')}] - {self.service_name} - log_worker - DB ERROR - {str(e)}")
        finally:
            session.close()
    
    def emit(self, record):
        try:
            if hasattr(record, 'module') and record.module in ['log_controller', 'log_handler', 'database']:
                return
            
            log_message = self.format(record)
            log_message = re.sub(r'^\[\d{4}-\d{2}-\d{2}T[\d\.:+\-]+\]', '', log_message).strip()
            
            if 'WebSocket' in log_message or 'socketio' in log_message:
                return
            
            log_fingerprint = hashlib.md5(f"{record.levelno}-{record.module}-{log_message}".encode('utf-8')).hexdigest()
            current_time = datetime.now().timestamp()
            
            if log_fingerprint in self.recent_logs:
                if current_time - self.recent_logs[log_fingerprint] < self.log_ttl:
                    return
            
            self.recent_logs[log_fingerprint] = current_time
            
            if len(self.recent_logs) > self.max_recent_logs:
                self.recent_logs = {fp: ts for fp, ts in self.recent_logs.items() if current_time - ts < self.log_ttl}
            
            log_data = {
                'time': datetime.now(timezone(timedelta(hours=8))),
                'level': record.levelname.upper(),
                'module': record.module if hasattr(record, 'module') else 'unknown',
                'category': getattr(record, 'category', 'system').lower(),
                'source': getattr(record, 'source', 'backend').lower(),
                'content': log_message,
                'task_id': getattr(record, 'task_id', None),
                'device_id': getattr(record, 'device_id', None),
                'api_id': getattr(record, 'api_id', None),
                'test_case_id': getattr(record, 'test_case_id', None),
                'thread_id': getattr(record, 'thread_id', None) or str(threading.get_ident()),
                'algorithm_type': getattr(record, 'algorithm_type', None),
                'service_name': getattr(record, 'service_name', self.service_name),
                'user_id': getattr(record, 'user_id', None),  # 新增
                'push_to_websocket': getattr(record, 'push_to_websocket', True),
            }
            
            self.queue.put(log_data)
            
        except Exception as e:
            if self.enable_console_log:
                print(f"Error putting log into queue: {str(e)}", file=sys.stderr)
```

```python
def get_task_logs(task_id):
    """获取任务日志（检查权限）"""
    from shared.models.models import TestTask
```
