# 测试任务执行服务拆分方案

## 1. 背景与目标

### 1.1 当前问题

当前系统为单体 Flask 应用，所有功能（路由、设备驱动、任务执行、评估计算）都在同一个进程中运行，存在以下问题：

| 问题 | 说明 |
|------|------|
| **耦合严重** | 设备驱动与 Flask HTTP Server 共用进程，设备通信阻塞 HTTP 响应 |
| **扩展性差** | 无法独立扩展任务执行能力 |
| **故障影响大** | 任务执行异常可能影响 API 响应 |
| **资源竞争** | 设备控制线程与 HTTP 处理线程竞争资源 |
| **部署不灵活** | 无法针对不同负载场景独立部署 |

### 1.2 拆分目标

1. **API Gateway** - 轻量级 HTTP 路由、认证、限流、响应格式化
2. **Task Service** - 任务调度、测试执行、评估计算、算法映射
3. **Device Service** - 设备驱动、通信控制、音频播放（SPL 映射）

### 5.2 Redis Pub/Sub 实现

```python
# shared/utils/redis_pubsub.py

import redis
import json
import threading
from datetime import datetime, timezone, timedelta

class RedisPubSub:
    def __init__(self, redis_url="redis://localhost:6379"):
        self.redis_url = redis_url
        self.redis_client = redis.from_url(redis_url)
        self.pubsub = self.redis_client.pubsub()

    def publish(self, channel, message):
        """发布消息"""
        if isinstance(message, dict):
            message = json.dumps(message)
        self.redis_client.publish(channel, message)

    def subscribe(self, channel, callback):
        """订阅频道"""
        self.pubsub.subscribe(channel)
        for message in self.pubsub.listen():
            if message['type'] == 'message':
                callback(json.loads(message['data']))


# 日志写入示例
class LogWriter:
    def __init__(self, service_name, redis_url):
        self.service_name = service_name
        self.pubsub = RedisPubSub(redis_url)

    def log(self, level, module, content, task_id=None, **kwargs):
        """写入日志"""
        log_entry = {
            'service': self.service_name,
            'time': datetime.now(timezone(timedelta(hours=8))).isoformat(),
            'level': level.upper(),
            'module': module,
            'content': content,
            'task_id': task_id,
            **kwargs
        }

        # 1. 写入数据库
        self._write_to_db(log_entry)

        # 2. 发布到 Redis
        self.pubsub.publish('task_logs', log_entry)

    def _write_to_db(self, log_entry):
        """写入 PostgreSQL"""
        # 使用独立数据库会话
        from shared.models.database import db
        log = Log(**log_entry)
        db.session.add(log)
        db.session.commit()
```

### 5.3 日志分库分表策略

当前系统运行不到一个月已有百万级日志，单表存储存在以下问题：

| 问题 | 影响 |
|------|------|
| 查询变慢 | `ORDER BY time DESC LIMIT 100` 全表扫描 |
| 存储成本 | 单表持续增长 |
| 归档困难 | 历史数据无法有效清理 |
| 索引膨胀 | B-tree 索引过大 |

#### 5.3.2 分区管理器

```python
# shared/utils/log_partition_manager.py

from datetime import datetime, timedelta
from sqlalchemy import text
import logging

class LogPartitionManager:
    """日志分区管理器 - 自动创建和维护分区"""

    def __init__(self, db_engine):
        self.engine = db_engine
        self.logger = logging.getLogger(__name__)

    def ensure_partition_exists(self, target_date=None):
        """确保目标月份的分区存在"""
        if target_date is None:
            target_date = datetime.now()

        # 计算下个月的分区
        if target_date.month == 12:
            next_month = target_date.replace(year=target_date.year + 1, month=1, day=1)
        else:
            next_month = target_date.replace(month=target_date.month + 1, day=1)

        partition_name = f'logs_{next_month.strftime("%Y_%m")}'

        with self.engine.connect() as conn:
            result = conn.execute(text(f"""
                SELECT EXISTS (
                    SELECT 1 FROM pg_tables
                    WHERE tablename = '{partition_name}'
                );
            """))

            if not result.scalar():
                self._create_partition(next_month)
                self.logger.info(f"Created partition: {partition_name}")

    def _create_partition(self, partition_date):
        """创建单个分区"""
        start_date = partition_date.strftime('%Y-%m-%d')
        end_date = (partition_date + timedelta(days=32)).replace(day=1).strftime('%Y-%m-%d')
        partition_name = f'logs_{partition_date.strftime("%Y_%m")}'

        with self.engine.connect() as conn:
            conn.execute(text(f"""
                CREATE TABLE IF NOT EXISTS {partition_name}
                PARTITION OF logs_partitioned
                FOR VALUES FROM ('{start_date}') TO ('{end_date}')
            """))

            conn.execute(text(f"""
                CREATE INDEX IF NOT EXISTS idx_{partition_name}_task_id
                ON {partition_name}(task_id)
            """))

            conn.execute(text(f"""
                CREATE INDEX IF NOT EXISTS idx_{partition_name}_time
                ON {partition_name}(time DESC)
            """))

            conn.execute(text(f"""
                CREATE INDEX IF NOT EXISTS idx_{partition_name}_service
                ON {partition_name}(service_name)
            """))

            conn.commit()
            self.logger.info(f"Created partition table: {partition_name}")

    def cleanup_old_partitions(self, retention_months=3):
        """清理旧分区（保留最近N个月）"""
        cutoff = datetime.now() - timedelta(days=retention_months * 30)
        cutoff_name = f'logs_{cutoff.strftime("%Y_%m")}'

        with self.engine.connect() as conn:
            result = conn.execute(text("""
                SELECT tablename FROM pg_tables
                WHERE tablename LIKE 'logs_%'
                AND schemaname = 'public'
            """))

            for row in result:
                partition_name = row[0]
                if partition_name < cutoff_name and partition_name.startswith('logs_20'):
                    self.logger.info(f"Dropping old partition: {partition_name}")
                    conn.execute(text(f"DROP TABLE IF EXISTS {partition_name}"))

            conn.commit()

    def get_partition_info(self):
        """获取所有分区信息"""
        with self.engine.connect() as conn:
            result = conn.execute(text("""
                SELECT
                    child.relname AS partition_name,
                    pg_size_pretty(pg_relation_size(child.oid)) AS size,
                    (SELECT COUNT(*) FROM logs_partitioned WHERE time >= splits[1] AND time < splits[2]) AS row_count
                FROM pg_inherits
                JOIN pg_class parent ON pg_inherits.inhparent = parent.oid
                JOIN pg_class child ON pg_inherits.inhrelid = child.oid
                WHERE parent.relname = 'logs_partitioned'
            """))
            return [dict(row) for row in result]
```

#### 5.3.3 自动分区检查

```python
# shared/services/log_service.py

from datetime import datetime
from contextlib import contextmanager
import threading

class LogService:
    """日志服务 - 带分区管理"""

    _partition_checked_today = False
    _partition_lock = threading.Lock()

    def __init__(self, db_engine, partition_manager):
        self.db = db_engine
        self.partition_manager = partition_manager

    def _ensure_partition(self):
        """确保分区存在（每天检查一次）"""
        with LogService._partition_lock:
            today = datetime.now().strftime('%Y%m%d')
            if not hasattr(LogService, '_last_check_date') or LogService._last_check_date != today:
                # 检查下个月分区是否存在
                self.partition_manager.ensure_partition_exists()
                LogService._last_check_date = today

    @contextmanager
    def session(self):
        """提供数据库会话"""
        session = self.db.session()
        try:
            yield session
            session.commit()
        except Exception:
            session.rollback()
            raise
        finally:
            session.close()

    def write_log(self, level, module, content, task_id=None, service_name=None, **kwargs):
        """写入日志"""
        self._ensure_partition()

        with self.session() as session:
            log = Log(
                level=level.upper(),
                module=module,
                content=content,
                task_id=task_id,
                service_name=service_name or 'unknown',
                time=datetime.now(),
                **kwargs
            )
            session.add(log)

    def query_logs(self, task_id=None, service=None, start_time=None, end_time=None, limit=100):
        """查询日志 - 自动利用分区裁剪"""
        self._ensure_partition()

        with self.session() as session:
            query = session.query(Log)

            if task_id:
                query = query.filter(Log.task_id == task_id)
            if service:
                query = query.filter(Log.service_name == service)
            if start_time:
                query = query.filter(Log.time >= start_time)
            if end_time:
                query = query.filter(Log.time < end_time)

            return query.order_by(Log.time.desc()).limit(limit).all()
```

```python
from apscheduler.schedulers.background import BackgroundScheduler
import logging

def setup_log_maintenance(db_engine):
    """设置日志维护定时任务"""

        trigger='cron',
        hour=2,
        minute=0,
        id='create_log_partition',
        name='创建日志分区',
        replace_existing=True

    scheduler.add_job(

        hour=3,
        minute=0,
        id='cleanup_old_partitions',
        name='清理旧日志分区',
        replace_existing=True
    )

    scheduler.start()
```

```python
import subprocess

import os

def run_service(name, module, port, env_vars=None):
```
