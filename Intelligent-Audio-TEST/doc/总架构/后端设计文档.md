# Intelligent Audio TEST 后端架构文档

## 1. 系统概述

Intelligent Audio TEST 是一个智能音频测试系统后端服务，提供测试用例管理、设备管理、任务执行、评估分析等功能。系统支持 API 测试和端到端（E2E）测试两种模式，基于 Flask + SQLAlchemy + PostgreSql 构建。

### 1.1 系统特性

| 特性 | 说明 |
|------|------|
| **双测试模式** | 支持 API 测试（高并发）和 E2E 端到端测试 |
| **实时进度** | WebSocket + SSE 双通道实时推送任务进度 |
| **多设备支持** | Android、HarmonyOS 等多类型设备驱动 |
| **算法配置化** | 支持 translation、asr、tts、speaker_recognition 等算法 |
| **自动调度** | 后台调度器自动启动 pending 状态任务 |
| **资源控制** | 线程池、任务队列、停止/暂停机制 |

### 1.2 技术栈

| 组件 | 技术选型 | 用途 |
|------|---------|------|
| Web 框架 | Flask 3.x | HTTP REST API |
| 数据库 | PostgreSQL 15+ + SQLAlchemy | 数据持久化 |
| **对象存储** | **MinIO / S3** | **音频、报告、日志存储** |
| 实时通信 | Flask-SocketIO + SSE | 进度推送、日志推送 |
| 任务调度 | APScheduler (内置) | 后台任务调度 |
| 音频处理 | pydub + ffmpeg | 音频格式转换、播放 |
| 数据验证 | Pydantic | 请求/响应数据校验 |
| CORS | flask-cors | 跨域支持 |

## 3. 架构分层详解

### 3.1 分层职责

## 4. 存储层设计

### 4.1 存储接口抽象

为支持本地存储和 S3 对象存储的灵活切换，设计存储接口：

```python
# storage/base.py
from abc import ABC, abstractmethod
from typing import BinaryIO, Optional
import datetime

class IStorage(ABC):
    """存储接口抽象"""

    @abstractmethod
    def save_audio(self, key: str, data: BinaryIO, metadata: dict = None) -> str:
        """保存音频文件"""
        pass

    @abstractmethod
    def save_report(self, key: str, data: BinaryIO, metadata: dict = None) -> str:
        """保存报告文件"""
        pass

    @abstractmethod
    def save_log(self, key: str, data: str, metadata: dict = None) -> str:
        """保存日志文件"""
        pass

    @abstractmethod
    def get_presigned_url(self, key: str, expires: int = 3600) -> str:
        """获取预签名下载 URL"""
        pass

    @abstractmethod
    def delete(self, key: str) -> bool:
        """删除文件"""
        pass

    @abstractmethod
    def exists(self, key: str) -> bool:
        """检查文件是否存在"""
        pass
```

### 4.2 本地存储实现

```python
# storage/local.py
import os
from pathlib import Path

class LocalStorage(IStorage):
    """本地文件系统存储"""

    def __init__(self, base_path: str = "./uploads"):
        self.base_path = Path(base_path)

    def _get_full_path(self, key: str) -> Path:
        return self.base_path / key

    def save_audio(self, key: str, data: BinaryIO, metadata: dict = None) -> str:
        path = self._get_full_path(f"audios/{key}")
        path.parent.mkdir(parents=True, exist_ok=True)
        with open(path, 'wb') as f:
            f.write(data.read())
        return str(path)

    def save_report(self, key: str, data: BinaryIO, metadata: dict = None) -> str:
        path = self._get_full_path(f"reports/{key}")
        path.parent.mkdir(parents=True, exist_ok=True)
        with open(path, 'wb') as f:
            f.write(data.read())
        return str(path)

    def get_presigned_url(self, key: str, expires: int = 3600) -> str:
        return f"file://{self._get_full_path(key)}"
```

### 4.3 MinIO/S3 存储实现

```python
# storage/s3.py
from minio import Minio
from datetime import timedelta

class S3Storage(IStorage):
    """S3 兼容对象存储 (MinIO / AWS S3)"""

    def __init__(self,
                 endpoint: str = "localhost:9000",
                 access_key: str = "minioadmin",
                 secret_key: str = "minioadmin",
                 bucket: str = "intelligent-audio",
                 secure: bool = False):
        self.client = Minio(
            endpoint,
            access_key=access_key,
            secret_key=secret_key,
            secure=secure
        )
        self.bucket = bucket
        self._ensure_bucket()

    def _ensure_bucket(self):
        if not self.client.bucket_exists(self.bucket):
            self.client.make_bucket(self.bucket)

    def save_audio(self, key: str, data: BinaryIO, metadata: dict = None) -> str:
        full_key = f"audios/{key}"
        self.client.put_object(
            self.bucket, full_key, data,
            length=data.getbuffer().nbytes,
            metadata=metadata or {}
        )
        return full_key

    def save_report(self, key: str, data: BinaryIO, metadata: dict = None) -> str:
        full_key = f"reports/{key}"
        self.client.put_object(
            self.bucket, full_key, data,
            length=data.getbuffer().nbytes,
            metadata=metadata or {}
        )
        return full_key

    def get_presigned_url(self, key: str, expires: int = 3600) -> str:
        return self.client.presigned_get_object(
            self.bucket, key,
            expires=timedelta(seconds=expires)
        )

    def delete(self, key: str) -> bool:
        self.client.remove_object(self.bucket, key)
        return True

    def exists(self, key: str) -> bool:
        try:
            self.client.stat_object(self.bucket, key)
            return True
        except:
            return False
```

### 4.4 存储配置

```python
# config/storage.py
from storage.local import LocalStorage
from storage.s3 import S3Storage

def get_storage():
    """根据配置获取存储实例"""
    import os

    storage_type = os.getenv("STORAGE_TYPE", "local")

    if storage_type == "s3":
        return S3Storage(
            endpoint=os.getenv("S3_ENDPOINT", "localhost:9000"),
            access_key=os.getenv("S3_ACCESS_KEY", "minioadmin"),
            secret_key=os.getenv("S3_SECRET_KEY", "minioadmin"),
            bucket=os.getenv("S3_BUCKET", "intelligent-audio"),
            secure=os.getenv("S3_SECURE", "false").lower() == "true"
        )
    else:
        return LocalStorage(
            base_path=os.getenv("STORAGE_PATH", "./uploads")
        )
```

### 7.1 单例模式

```python
class ExecutionEngine:
    """任务执行引擎 - 单例模式"""
    _instance = None
    _lock = threading.Lock()

    def __new__(cls):
        if cls._instance is None:
            with cls._lock:
                if cls._instance is None:
                    cls._instance = super().__new__(cls)
                    cls._instance.workers = {}
                    cls._instance.stop_flags = {}
                    cls._instance.pause_flags = {}
        return cls._instance

execution_engine = ExecutionEngine()
```

### 7.2 线程池管理

```python
class ExecutionEngine:
    def _run_task(self, app, task_id, ...):
        if task.type == 'api':
            max_workers = sum(ep.get('max_process', 5)
                            for ep in available_endpoints)
            self.api_executors[task_id] = ThreadPoolExecutor(
                max_workers=max_workers
            )
            self.api_executors[task_id].submit(
                self._execute_api_case, app, task_id, tc_rel_id
            )
        else:
            self._execute_e2e_case(task_id, tc_rel_id)  # 同步
```

### 7.3 后台调度器

```python
class ExecutionEngine:
    def _scheduler_loop(self):
        while not self.scheduler_stop_event.is_set():
            self._schedule_pending_tasks()
            self.scheduler_stop_event.wait(timeout=check_interval)

    def _schedule_pending_tasks(self):
        pending_tasks = Task.query.filter_by(status='pending').all()
        for task in pending_tasks:
            if can_run(task):
                self.start_task(app, task.id)
```

### 7.4 资源控制

```python
class ExecutionEngine:
    def __init__(self):
        self.running_apis = set()      # API 互斥
        self.running_e2e = False       # E2E 全局锁

    def can_run_task(self, task):
        if task.type == 'e2e':
            return not self.running_e2e
        else:
            overlapping = set(task.api_ids) & self.running_apis
            return len(overlapping) == 0
```

### 8.1 ConfigManager

```python
class ConfigManager:
    """配置管理器 - 单例模式"""
    _instance = None

    def __new__(cls):
        if cls._instance is None:
            with cls._lock:
                cls._instance = super().__new__(cls)
                cls._instance.config = {}
                cls._instance.load_config()
        return cls._instance

    def load_config(self):
        """从 JSON 文件加载配置"""
        with open(self.config_path, 'r', encoding='utf-8') as f:
            self.config = json.load(f)

    def get_value(self, service, key, default=None):
        """获取配置值，支持 Flask 配置覆盖"""
        value = self.config.get(service, {}).get(key)
        if value is None and self.flask_config:
            flask_key = f"{service.upper()}_{key.upper()}"
            value = self.flask_config.get(flask_key)
        return value if value is not None else default

config_manager = ConfigManager()
```

                    │
              EvaluationService

**队列设计**:
